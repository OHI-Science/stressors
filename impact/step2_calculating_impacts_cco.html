# write_csv(avg_hab_impact_tmp, here::here('impact/int/avg_hab_impact_tmp.csv'))
calc_start_end <- function(df, group = 'habitat', var = 'pressure', val = 'impact') {
  ordered_df <- df %>%
    ### flag negative vars
    mutate(neg = !!sym(val) < 0) %>%
    ### arrange by group (e.g., habitat, region) and by a particular variable (e.g., pressure)
    ### move negative values to the front by ordering by `neg` flag
    arrange(!!sym(group), -neg, !!sym(var))
  
  out_df <- ordered_df %>%
    group_by(!!sym(group)) %>%
    ### deal with negs by flipping to positive then nudging negative after
    ### first, set up temp variable to hold the value of interest; find the total
    ### negative contributions; then flip the values to positive if neg and do 
    ### cumulative sums; then nudge the entire cum sums by the total negative contributions
    mutate(tmp = !!sym(val),
           nudge = sum(tmp[tmp < 0]),
           tmp = ifelse(neg, -tmp, tmp),
           end = cumsum(tmp),
           start = lag(end, default = 0) + nudge,
           end = end + nudge) %>%
    ungroup() %>%
    select(-neg, -tmp, -nudge)
  
  return(out_df)
}

## make data wide
avg_hab_impact <- avg_hab_impact_tmp %>%
  pivot_wider(
    names_from = scen_time,
    values_from = avg_impact,
    values_fill = list(avg_impact = 0)) %>%
  mutate(ssp245_change = `ssp245_medium-term`-ssp245_current,
         ssp585_change = `ssp585_medium-term`-ssp585_current)

# Step 1: Summarize the data to get total values for each pressure within each habitat
pressure_change_order <- avg_hab_impact %>%
  group_by(pressure) %>%
  summarize(total_value = sum(`ssp245_change`), .groups = 'drop') %>%
  arrange(total_value) %>%
  mutate(pressure = factor(pressure, levels = unique(pressure)))


habitat_change_order <- avg_hab_impact %>%
  group_by(habitat) %>%
  summarize(total_value = sum(`ssp245_change`), .groups='drop') %>%
  arrange(desc(total_value)) %>%
  mutate(habitat = factor(habitat, levels = unique(habitat)))


 # Step 2: Get plot data and reorder the 'pressure' factor levels based on the total_value
future_plot_data <- avg_hab_impact %>%
  select(habitat, pressure, impact =`ssp245_medium-term`, color) %>%
  mutate(pressure = factor(pressure, levels = unique(pressure_change_order$pressure))) %>%
  mutate(habitat = factor(habitat, levels = unique(habitat_change_order$habitat))) %>%
  arrange(habitat, pressure) %>%
  calc_start_end(group = 'habitat')

current_plot_data <- avg_hab_impact %>%
  select(habitat, pressure, impact = ssp245_current, color) %>%
  mutate(pressure = factor(pressure, levels = unique(pressure_change_order$pressure))) %>%
  mutate(habitat = factor(habitat, levels = unique(habitat_change_order$habitat))) %>%
  calc_start_end(group = 'habitat')

current_totals <- current_plot_data %>%
